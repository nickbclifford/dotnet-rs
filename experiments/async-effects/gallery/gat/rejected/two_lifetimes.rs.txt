pub trait NativeHost: 'static {
    type Ctx<'gc, 'a>: RegionAccess<'gc> where 'gc: 'a;
    fn enter<R>(&mut self, f: impl for<'gc, 'a> FnOnce(&mut Self::Ctx<'gc, 'a>) -> R) -> R;
    fn perform(&mut self, effect: Effect) -> impl Future<Output = Outcome>;
}
impl NativeHost for VmEffects {
    type Ctx<'gc, 'a> = Region<'gc, 'a> where 'gc: 'a;
    fn enter<R>(&mut self, f: impl for<'gc, 'a> FnOnce(&mut Self::Ctx<'gc, 'a>) -> R) -> R { self.enter(f) }
    async fn perform(&mut self, effect: Effect) -> Outcome { self.perform(effect).await }
}
