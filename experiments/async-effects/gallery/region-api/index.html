D is selected: let mut vm = fx.vm().await?, an RAII access loan lending rooted views after honoring a pending stop. Read the decision and safety contract.

All four columns compile for the same seven operations. A/B/C are retained alternatives. The previous always-ready callback uses vm_ready; vm() is reserved for the selected RAII API. The rejected async-closure probe is covered separately in tests.

pub async fn substring(
    mut fx: VmEffects,
    source: StringRoot,
    start: i32,
    length: Option<i32>,
) -> Outcome {
    let count = fx.with_vm(|c| {
        let source = source.get(c);
        let source = source.borrow();
        let ObjectData::String(chars) = &*source else {
            return Err(Failure::Managed("ArgumentException"));
        };
        if start < 0 || start as usize > chars.len() {
            return Err(Failure::Managed("ArgumentOutOfRangeException:startIndex"));
        }
        let count = length.unwrap_or((chars.len() - start as usize) as i32);
        if count < 0 || count as usize > chars.len() - start as usize {
            return Err(Failure::Managed("ArgumentOutOfRangeException:length"));
        }
        Ok(count as usize)
    })?;
    let mut result = Vec::with_capacity(count);
    let mut offset = 0;
    const CHUNK_SIZE: usize = 1024 * 64;
    while offset < count {
        let end = (offset + CHUNK_SIZE).min(count);
        let requested = fx.with_vm(|c| {
            let source = source.get(c);
            {
                let source = source.borrow();
                let ObjectData::String(chars) = &*source else {
                    unreachable!()
                };
                result.extend_from_slice(&chars[start as usize + offset..start as usize + end]);
            }
            c.host().copied(end - offset);
            c.requested()
        });
        offset = end;
        if offset < count && requested {
            fx.safe_point().await?;
        }
    }
    Ok(fx.with_vm(|c| c.string(result).0))
}

pub async fn substring(
    mut fx: VmEffects,
    source: StringRoot,
    start: i32,
    length: Option<i32>,
) -> Outcome {
    let count = {
        let mut vm = fx.vm().await?;
        vm.string(&source).substring_len(start, length)?
    };
    let mut result = Vec::with_capacity(count);
    let mut offset = 0;
    const CHUNK_SIZE: usize = 1024 * 64;
    while offset < count {
        let end = (offset + CHUNK_SIZE).min(count);
        {
            let mut vm = fx.vm().await?;
            vm.string(&source).copy_range(
                start as usize + offset,
                start as usize + end,
                &mut result,
            )?;
        }
        offset = end;
    }
    let mut vm = fx.vm().await?;
    Ok(vm.new_string(result))
}

pub async fn reflection_invoke(mut fx: VmEffects, spec: InvokeSpec) -> Outcome {
    let call = fx.with_vm(|c| prepare_invoke(c, &spec))?;
    let result = fx.call_managed(call).await;
    // Byref writes are through owned cells. A transparent throw unwinds this native frame;
    // normal completion alone reaches return boxing, matching the existing remote hook.
    fx.with_vm(|c| finish_invoke(c, &spec, result))
}

pub async fn ensure_static(mut fx: VmEffects, ty: usize) -> Outcome {
    loop {
        match fx.with_vm(|c| c.host().static_action(ty)) {
            StaticAction::Ready => return Ok(Stashed::Void),
            StaticAction::Failed => return Err(Failure::Managed("TypeInitializationException")),
            StaticAction::Run(method) => {
                // VM marks initialized/failed on .cctor frame retirement, as today.
                fx.call_managed(Call {
                    propagation: Propagation::Transparent,
                    method,
                    args: vec![],
                    byrefs: vec![],
                })
                .await?;
            }
            StaticAction::Wait => {
                fx.wait_static_init(ty).await?;
            }
        }
    }
}
